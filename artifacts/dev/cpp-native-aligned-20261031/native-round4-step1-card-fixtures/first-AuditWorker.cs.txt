using System.Reflection;
using System.Security.Cryptography;
using System.Text.Json;
using Godot;
using MegaCrit.Sts2.Core.Combat;
using MegaCrit.Sts2.Core.Commands;
using MegaCrit.Sts2.Core.Entities.Cards;
using MegaCrit.Sts2.Core.Entities.Creatures;
using MegaCrit.Sts2.Core.GameActions.Multiplayer;
using MegaCrit.Sts2.Core.Helpers;
using MegaCrit.Sts2.Core.Hooks;
using MegaCrit.Sts2.Core.Models;
using MegaCrit.Sts2.Core.Models.Powers;
using MegaCrit.Sts2.Core.Models.Relics;
using MegaCrit.Sts2.Core.TestSupport;
using MegaCrit.Sts2.Core.ValueProps;

public partial class AuditWorker : Node
{
    readonly List<object> results = [];
    public override async void _Ready()
    {
        string output = System.Environment.GetEnvironmentVariable("STS2_AUDIT_OUT")!;
        try
        {
            TestMode.IsOn = true;
            NonInteractiveMode.AutoSlayerCheck = () => true;
            MegaCrit.Sts2.Core.Context.LocalContext.NetId = 1;
            if (!ProjectSettings.LoadResourcePack(System.Environment.GetEnvironmentVariable("STS2_GAME_PACK")!, false)) throw new IOException("Game pack failed to load");
            await OneTimeInitialization.ExecuteVeryEarly();
            var saves = MegaCrit.Sts2.Core.Saves.SaveManager.Instance;
            saves.InitPrefsDataForTest(); saves.InitProfileId();
            OneTimeInitialization.ExecuteEssential(); saves.InitProgressData();
            typeof(NativeSession).Assembly.GetType("NativeReplayCompatibilityPatches")!.GetMethod("Install", BindingFlags.Static | BindingFlags.Public)!.Invoke(null, null);
            CombatSolver.Api.NativeMctsSimulationApi.Initialize();
            SlayTheModel.Sts2.ModAdapter.NativeCombatCheckpoint.CaptureEnabled = false;
            using var timeout = new CancellationTokenSource(TimeSpan.FromSeconds(90));
            var session = new NativeSession(this) { ChoiceFixture = true };
            var choice = new ThrowingPlayerChoiceContext();
            async Task Reset(string[] ids, bool upgraded=false) {
                session.ChoiceFixtureCards=ids.Concat(new[]{"DEFEND_IRONCLAD"}).ToArray();
                session.ChoiceFixtureUpgradedCards=upgraded?new HashSet<string>(ids):[];
                await session.ResetAsync("NATIVE-ROUND4-STEP1",timeout.Token);
                foreach(var enemy in session.CombatStateForSimulation.Enemies) { enemy.SetMaxHpInternal(10000); enemy.SetCurrentHpInternal(10000); }
                var player=session.CombatStateForSimulation.Players[0];
                player.Creature.SetMaxHpInternal(80);player.Creature.SetCurrentHpInternal(80);
                await PlayerCmd.SetEnergy(10,player);
            }
            async Task Play(string id,int count) {
                var pc=session.CombatStateForSimulation.Players[0].PlayerCombatState!;
                pc.Hand.Cards.First(c=>c.Id.Entry==id).BaseReplayCount=count-1;
                await session.StepAsync(session.ActionForCard(id),timeout.Token);
            }
            async Task Modifier(string modifier) {
                var player=session.CombatStateForSimulation.Players[0];var creature=player.Creature;
                await PowerCmd.Apply<RupturePower>(choice,creature,2,creature,null);
                if(modifier.Contains("tungsten")) await RelicCmd.Obtain<TungstenRod>(player);
                if(modifier.Contains("buffer")) await PowerCmd.Apply<BufferPower>(choice,creature,1,creature,null);
                if(modifier=="intangible") await PowerCmd.Apply<IntangiblePower>(choice,creature,1,creature,null);
                if(modifier=="revive") {await RelicCmd.Obtain<LizardTail>(player);creature.SetCurrentHpInternal(1);}
                if(modifier=="lethal") creature.SetCurrentHpInternal(1);
            }
            async Task Turn() {
                var combat=session.CombatStateForSimulation;var player=combat.Players[0];
                player.Creature.LoseBlockInternal(player.Creature.Block);
                await Hook.AfterPlayerTurnStart(combat,choice,player);
            }
            object State() {
                var combat=session.CombatStateForSimulation;var player=combat.Players[0];var c=player.Creature;
                return new {hp=c.CurrentHp,block=c.Block,strength=c.GetPowerAmount<StrengthPower>(),buffer=c.GetPowerAmount<BufferPower>(),
                    crimsonAmount=c.GetPowerAmount<CrimsonMantlePower>(),infernoAmount=c.GetPowerAmount<InfernoPower>(),
                    crimsonSelf=c.GetPower<CrimsonMantlePower>()?.DynamicVars["SelfDamage"].BaseValue??0,
                    infernoSelf=c.GetPower<InfernoPower>()?.DynamicVars["SelfDamage"].BaseValue??0,
                    powers=c.Powers.Select(p=>p.Id.Entry).ToArray(),damage=combat.Enemies.Select(e=>10000-e.CurrentHp).ToArray(),
                    lizardUsed=player.Relics.OfType<LizardTail>().FirstOrDefault()?.WasUsed??false};
            }
            foreach(string id in new[]{"CRIMSON_MANTLE","INFERNO"})
            foreach(bool upgraded in new[]{false,true})
            foreach(int count in new[]{1,2,3}) {
                await Reset([id],upgraded);await Play(id,count);await Modifier("none");
                var before=State();await Turn();
                var first=State();await Turn();
                results.Add(new{test="stack",id,upgraded,count,before,first,second=State()});
            }
            foreach(string id in new[]{"CRIMSON_MANTLE","INFERNO"})
            foreach(string modifier in new[]{"tungsten","buffer","tungsten_buffer","intangible","revive","lethal"}) {
                await Reset([id]);await Play(id,3);await Modifier(modifier);
                var before=State();await Turn();
                results.Add(new{test="modifier",id,modifier,before,after=State()});
            }
            foreach(bool infernoFirst in new[]{false,true})
            foreach(string modifier in new[]{"none","buffer","tungsten","intangible","revive"}) {
                await Reset(["CRIMSON_MANTLE","INFERNO"]);
                await Play(infernoFirst?"INFERNO":"CRIMSON_MANTLE",infernoFirst?3:1);
                await Play(infernoFirst?"CRIMSON_MANTLE":"INFERNO",infernoFirst?1:3);
                await Modifier(modifier);
                var before=State();await Turn();
                results.Add(new{test="order",infernoFirst,modifier,before,after=State()});
            }
            foreach(string mode in new[]{"own_damage","enemy_damage","direct_hp","blocked"}) {
                await Reset(["INFERNO"]);await Play("INFERNO",2);await Modifier("none");
                var combat=session.CombatStateForSimulation;var player=combat.Players[0];
                if(mode=="enemy_damage") combat.CurrentSide=CombatSide.Enemy;
                if(mode=="direct_hp") player.Creature.LoseHpInternal(3,ValueProp.Unblockable|ValueProp.Unpowered);
                else {
                    if(mode=="blocked") await CreatureCmd.GainBlock(player.Creature,10,ValueProp.Unpowered,null);
                    await CreatureCmd.Damage(choice,player.Creature,3,mode=="blocked"?ValueProp.Unpowered:ValueProp.Unblockable|ValueProp.Unpowered,null);
                }
                results.Add(new{test="damage_trigger",mode,after=State()});
            }
            foreach(string id in new[]{"CRIMSON_MANTLE","INFERNO"}) {
                await Reset([id]);var c=session.CombatStateForSimulation.Players[0].Creature;
                if(id=="CRIMSON_MANTLE") await PowerCmd.Apply<CrimsonMantlePower>(choice,c,13,c,null);
                else await PowerCmd.Apply<InfernoPower>(choice,c,13,c,null);
                await Turn();
                results.Add(new{test="direct_apply",id,after=State()});
                await Play(id,2);
                var before=State();
                if(id=="CRIMSON_MANTLE") await PowerCmd.Remove<CrimsonMantlePower>(c);else await PowerCmd.Remove<InfernoPower>(c);
                if(id=="CRIMSON_MANTLE") await PowerCmd.Apply<CrimsonMantlePower>(choice,c,13,c,null);else await PowerCmd.Apply<InfernoPower>(choice,c,13,c,null);
                results.Add(new{test="remove_reapply",id,before,after=State()});
            }
            File.WriteAllText(output, JsonSerializer.Serialize(new {status="completed",gameAssemblySha256=Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(typeof(ModelDb).Assembly.Location))).ToLowerInvariant(),gameSceneLoaded=MegaCrit.Sts2.Core.Nodes.NGame.Instance!=null,userDirectory=OS.GetUserDataDir(),scope="Actual card OnPlay/replay and controlled original turn hooks. Not full runs or general hook order parity. No simulator rule patches.",results},new JsonSerializerOptions{WriteIndented=true}));
            GD.Print("SIM_AUDIT_COMPLETED "+output);GetTree().Quit();
        }
        catch(Exception e) { File.WriteAllText(output,JsonSerializer.Serialize(new{status="error",error=e.ToString(),results},new JsonSerializerOptions{WriteIndented=true}));GD.PrintErr(e);GetTree().Quit(1); }
    }
}
